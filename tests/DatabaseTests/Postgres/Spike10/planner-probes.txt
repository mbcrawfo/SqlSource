-- The probes of the names the planner makes, run with psql on postgres:14.24, 15.19, 16.15, 17.11 and 18.6.
-- probe.sql
create table t (id int primary key, g int, v text unique);
create table c (id int primary key, parent int);
create table p (id int, k int) partition by range (k);
create table p1 partition of p for values from (0) to (10);
create table p2 partition of p for values from (10) to (20);
create table inh_parent(a int); create table inh_child() inherits (inh_parent);
\set QUIET on
\o /dev/null
\o
\pset format unaligned
\pset tuples_only on
\echo ###Q unnamed_subquery
explain (verbose, format json) select * from (select id from t offset 0);
\echo ###Q no-alias-subquery
explain (verbose, format json) select id from (select id from t offset 0) as q;
\echo ###Q excluded
explain (verbose, format json) insert into t values (1,2,'a')
  on conflict (id) do update set v = excluded.v
  where excluded.g > t.g;
\echo ###Q values
explain (verbose, format json) select * from (values (1,'a'),(2,'b')) as x(a,b), (values (1),(2)) y;
\echo ###Q recursive
explain (verbose, format json) with recursive r(n) as (select 1
  union all
  select n+1
  from r
  where n < 3)
  select *
  from r;
\echo ###Q any
explain (verbose, format json) select * from t where id in (select parent from c) or g = any (array[1,2]);
\echo ###Q initplan
explain (verbose, format json) select (select max(id) from c), id from t where g > (select min(id) from c);
\echo ###Q cte
explain (verbose, format json) with w as (select id from t) select * from w a join w b on a.id=b.id;
\echo ###Q multiexpr
explain (verbose, format json) update t set (g, v) = (select 1, 'x') where id = 1;
\echo ###Q setop
explain (verbose, format json) select id from t union select id from c except select parent from c;
\echo ###Q window-named
explain (verbose, format json)
  select row_number() over w, sum(g) over (partition by id), rank() over (order by g)
  from t
  window w as (order by id);
\echo ###Q partition-alias
explain (verbose, format json) select * from p x join inh_parent y on x.id=y.a;
\echo ###Q alias-dup
explain (verbose, format json)
  select *
  from t a
  join t a2 on true
  join c a3 on true
  join (select 1 as z offset 0) a on true;
explain (verbose, format json) select * from t a join t a2 on true join c a3 on true join (select 1) on true;
\echo ###Q columns-dup
explain (verbose, format json) select * from (select id, id, 1, 2 from t offset 0) q(a, b, c, d);
\echo ###Q join-alias
explain (verbose, format json) select * from t join c using (id) full join c c2 using (id);
\echo ###Q exists-anti
explain (verbose, format json)
  select *
  from t
  where not exists (select 1
  from c
  where c.parent = t.id)
  and exists (select 1
  from c);
\echo ###Q returning-oldnew
explain (verbose, format json) update t set g = g + 1 returning old.g, new.g;
\echo ###Q merge
explain (verbose, format json) merge into t
  using c on t.id = c.id
  when matched then update set g = 1
  when not matched then insert values (c.id, 1, 'm')
  when not matched by source then delete;
\echo ###Q grouping
explain (verbose, format json) select g, grouping(g), count(*) from t group by rollup(g);
\echo ###Q tablesample-func
explain (verbose, format json) select * from generate_series(1,3) gs, unnest(array[1,2]) with ordinality u(x,o);
\echo ###Q row-cols
explain (verbose, format json)
  select row(id, g), coalesce(g, 1), g::text, case
  when g=1 then 1 end, greatest(g, id), array[g], exists(select 1), g is null
  from t;

-- probe2.sql
create table t (id int primary key, g int, v text unique);
create table c (id int primary key, parent int);
\pset format unaligned
\pset tuples_only on
\echo ###Q unnamed-subscan
explain (verbose, format json) select * from (select id, random() r from t offset 0) where r > .5;
\echo ###Q any-offset
explain (verbose, format json) select * from t where id in (select parent from c offset 0) and g > 2;
\echo ###Q any-limit-join
explain (verbose, format json) select * from t where id in (select parent from c limit 5);
\echo ###Q any-nonhash
explain (verbose, format json) select * from t where (id, g) in (select parent, id from c where parent > t.g);
\echo ###Q join-using-full
explain (verbose, format json) select id from t full join c using (id) where id > 3;
\echo ###Q lateral-func
explain (verbose, format json)
  select *
  from t, lateral (select *
  from c
  where c.parent = t.id offset 0) l
  where l.id > 1;
\echo ###Q two-values
explain (verbose, format json)
  select *
  from (values (1),(2)) a(x)
  join (values (3)) as q on true, (values (4),(5)) as b(y);
\echo ###Q values-no-alias
explain (verbose, format json) select * from t where id in (values (1),(2));
\echo ###Q values-twice-noalias
explain (verbose, format json) insert into c
  select *
  from (values (1,2),(3,4)) v(id,parent)
  where random() > 0.5
  union all
  select *
  from (values (5,6)) w(id,parent);
